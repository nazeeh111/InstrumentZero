cmake_minimum_required(VERSION 3.20.0)
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(instrumentzero_edge)
if(NOT DEFINED IZ_BINDINGS_DIR)
  message(FATAL_ERROR "Generate reviewed bindings with instrumentzero edge compile SPEC --output DIR and pass -DIZ_BINDINGS_DIR=DIR")
endif()
target_include_directories(app PRIVATE ${IZ_BINDINGS_DIR} ../../src/instrumentzero/edge/core)
target_compile_options(app PRIVATE -fstack-usage)
target_sources(app PRIVATE src/main.c ../../src/instrumentzero/edge/core/iz_core.c)
